#!/bin/zsh

# Script to install Java 17 without sudo using SDKMAN!
# This script can be run by any user without root privileges

set -e  # Exit on error

echo "================================================"
echo "Java 17 Installation Script (No sudo required)"
echo "================================================"
echo ""

# Check if SDKMAN is already installed
if [[ -s "$HOME/.sdkman/bin/sdkman-init.sh" ]]; then
    echo "✓ SDKMAN! is already installed"
else
    echo "Installing SDKMAN!..."
    curl -s "https://get.sdkman.io" | bash
    echo ""
    echo "✓ SDKMAN! installed successfully"
fi

# Source SDKMAN
source "$HOME/.sdkman/bin/sdkman-init.sh"

echo ""
echo "Installing Java 17..."
sdk install java 17.0.9-tem

echo ""
echo "Setting Java 17 as default..."
sdk default java 17.0.9-tem

echo ""
echo "================================================"
echo "Installation Complete!"
echo "================================================"
echo ""
echo "Java version:"
java -version

echo ""
echo "JAVA_HOME is set to:"
echo $JAVA_HOME

echo ""
echo "================================================"
echo "Important: To use Java in new terminal sessions,"
echo "make sure the following is in your ~/.zshrc:"
echo ""
echo "export SDKMAN_DIR=\"\$HOME/.sdkman\""
echo "[[ -s \"\$HOME/.sdkman/bin/sdkman-init.sh\" ]] && source \"\$HOME/.sdkman/bin/sdkman-init.sh\""
echo "================================================"

# Add to .zshrc if not already present
if ! grep -q "sdkman-init.sh" "$HOME/.zshrc" 2>/dev/null; then
    echo ""
    echo "Adding SDKMAN initialization to ~/.zshrc..."
    cat >> "$HOME/.zshrc" << 'EOF'

# SDKMAN configuration
export SDKMAN_DIR="$HOME/.sdkman"
[[ -s "$HOME/.sdkman/bin/sdkman-init.sh" ]] && source "$HOME/.sdkman/bin/sdkman-init.sh"
EOF
    echo "✓ Added to ~/.zshrc"
else
    echo ""
    echo "✓ SDKMAN already configured in ~/.zshrc"
fi

echo ""
echo "You can now use Java 17!"
echo "Run 'source ~/.zshrc' or restart your terminal to ensure everything is loaded."
